import { Component } from '@angular/core';
import { MargeVente } from '../../model/marge-vente';
import { Subject, takeUntil } from 'rxjs';
import { MargeVenteFilter } from '../../model/marge-vente-filter';
import { FormBuilder, FormGroup, FormsModule, ReactiveFormsModule, Validators } from '@angular/forms';
import { MargeVenteServiceService } from '../../service/marge-vente-service.service';
import { MargeVenteStats } from '../../model/marge-vente-stats';
import { Annee } from '../../model/annee';
import { Depot } from '../../model/depot';
import { Mois } from '../../model/mois';
import { CommonModule } from '@angular/common';
import { UserService } from '../../service/user-service.service';


@Component({
  selector: 'app-marge-vente',
  standalone: true,
  imports: [CommonModule,
    ReactiveFormsModule,
    FormsModule
  ],
  templateUrl: './marge-vente.component.html',
  styleUrl: './marge-vente.component.css'
})
export class MargeVenteComponent {

  private destroy$ = new Subject<void>();

  margeVenteForm!: FormGroup;
  editForm!: FormGroup;

  // Données
  margesVente: MargeVente[] = [];
  annees: Annee[] = [];
  moisByAnnee: Mois[] = [];
  datesByMois: Date[] = [];
  depots: Depot[] = [];

  // États
  position = 1; // 1 = jour, 2 = mois
  isLoading = false;
  selectedMarge: MargeVente | null = null;
  showEditModal = false;
  usertupdate!: string;

  // Statistiques
  stats: MargeVenteStats = {
    totalCaisse: 0,
    totalAchat: 0,
    remise: 0,
    totalMarge: 0
  };

  // Configuration du tableau
  tableConfig = {
    currentPage: 1,
    pageSize: 50,
    totalRecords: 0,
    pageSizeOptions: [5, 10, 20, 50, 100, 500, 1000]
  };

  constructor(
    private fb: FormBuilder,
    private margeVenteService: MargeVenteServiceService, private userService: UserService
  ) {
    this.initializeForms();
  }

  ngOnInit(): void {
    this.loadInitialData();
    this.subscribeToStats();
    this.usertupdate = this.userService.getUserConnected().username;
  }

  ngOnDestroy(): void {
    this.destroy$.next();
    this.destroy$.complete();
  }

  private initializeForms(): void {
    this.margeVenteForm = this.fb.group({
      annee: [null, Validators.required],
      mois: [null, Validators.required],
      date: [null],
      debut: [null],
      fin: [null],
      depot: [null, Validators.required]
    });

    this.editForm = this.fb.group({
      achat: [0, [Validators.required, Validators.min(0)]],
      prixVente: [0, [Validators.required, Validators.min(0)]]
    });
  }

  private loadInitialData(): void {
    this.isLoading = true;

    this.margeVenteService.getAnnees()
      .pipe(takeUntil(this.destroy$))
      .subscribe(annees => {
        this.annees = annees;
        this.isLoading = false;
      });

    this.margeVenteService.getDepots()
      .pipe(takeUntil(this.destroy$))
      .subscribe(depots => {
        this.depots = depots;
      });
  }

  private subscribeToStats(): void {
    this.margeVenteService.stats$
      .pipe(takeUntil(this.destroy$))
      .subscribe(stats => {
        this.stats = stats;
      });
  }

  // Gestion des événements de formulaire
  onAnneeChange(): void {
    const annee = this.margeVenteForm.get('annee')?.value;
    if (annee) {
      this.margeVenteService.getDateByAnnee(annee.id)
        .pipe(takeUntil(this.destroy$))
        .subscribe(date => {
          this.datesByMois = date;
          //this.margeVenteForm.patchValue({ mois: null, date: null });
        });
    }
  }

  onMoisChange(): void {
    const mois = this.margeVenteForm.get('mois')?.value;
    if (mois && this.position === 1) {
      /**this.margeVenteService.getDatesByMois(mois.id)
        .pipe(takeUntil(this.destroy$))
        .subscribe(dates => {
          this.datesByMois = dates;
          this.margeVenteForm.patchValue({ date: null });
        });**/
    }
  }

  onPositionChange(): void {
    this.margeVenteForm.patchValue({ date: null });
    //this.datesByMois = [];
    this.margesVente = [];
    this.stats.remise=0;
    this.stats.totalAchat=0;
    this.stats.totalCaisse=0;
    this.stats.totalMarge=0;

  }

  onDateChange(): void {
    if (this.position === 1) {
      console.log("position", this.position)
      this.genererMarge();
    }
     if (this.position === 2) {
      console.log("position", this.position)
      this.genererMargeMensuel();
    }
    //this.operation();
  }

  // Actions principales
  genererMarge(): void {
    if (this.margeVenteForm.get('date')) {
      this.margesVente=[];
      this.isLoading = true;

      const filter: MargeVenteFilter = {
        annee: this.margeVenteForm.get('annee')?.value,
        // mois: this.margeVenteForm.get('mois')?.value,
        date: this.margeVenteForm.get('date')?.value,
        depot: this.margeVenteForm.get('depot')?.value,
        debut: new Date,
        fin:new Date,
        position: this.position
      };

      this.margeVenteService.getMargeJournaliere(filter.date ?? new Date, filter.annee?.id ?? 0)
        .pipe(takeUntil(this.destroy$))
        .subscribe({
          next: (marges) => {
            this.margesVente = marges;
            this.operation(marges);
            this.tableConfig.totalRecords = marges.length;
            this.isLoading = false;
          },
          error: (error) => {
            console.error('Erreur lors du chargement des marges:', error);
            this.isLoading = false;
          }
        });
    }
  }

   genererMargeMensuel(): void {
    if (this.margeVenteForm.get('debut') && this.margeVenteForm.get('fin') )  {
      this.isLoading = true;

      const filter: MargeVenteFilter = {
        annee: this.margeVenteForm.get('annee')?.value,
        // mois: this.margeVenteForm.get('mois')?.value,
        debut: this.margeVenteForm.get('debut')?.value,
        fin: this.margeVenteForm.get('fin')?.value,
        position: this.position
      };

      this.margeVenteService.getMargeMargeMensuel(filter.debut ?? new Date,filter.fin ?? new Date, filter.annee?.id ?? 0)
        .pipe(takeUntil(this.destroy$))
        .subscribe({
          next: (marges) => {
            this.margesVente = marges;
            this.operation(marges);
            this.tableConfig.totalRecords = marges.length;
            this.isLoading = false;
          },
          error: (error) => {
            console.error('Erreur lors du chargement des marges:', error);
            this.isLoading = false;
          }
        });
    }
  }

  rafraichir(): void {
    this.margeVenteForm.reset();
    this.margesVente = [];
    this.selectedMarge = null;
    this.position = 1;
    this.stats = {
      totalCaisse: 0,
      totalAchat: 0,
      remise: 0,
      totalMarge: 0
    };
  }

  editerMarge(marge: MargeVente): void {
    this.selectedMarge = { ...marge };
    this.editForm.patchValue({
      achat: marge.achat,
      prixVente: marge.prixVente
    });
    this.showEditModal = true;
  }

  validerEdition(): void {
    if (this.editForm.valid && this.selectedMarge) {
      const updatedMarge: MargeVente = {
        ...this.selectedMarge,
        achat: this.editForm.get('achat')?.value,
        prixVente: this.editForm.get('prixVente')?.value,
        usercreat: this.usertupdate,
        datevente: this.margeVenteForm.get('date')?.value
      };

      this.margeVenteService.validerAchat(updatedMarge)
        .pipe(takeUntil(this.destroy$))
        .subscribe({
          next: (result) => {
            this.showEditModal = false;
            this.genererMarge(); // Recharger les données
          },
          error: (error) => {
            console.error('Erreur lors de la validation:', error);
          }
        });
    }
  }

  annulerEdition(): void {
    this.showEditModal = false;
    this.selectedMarge = null;
    this.editForm.reset();
  }

  imprimerRapport(): void {
    if (this.margeVenteForm.get('annee')?.value && this.margeVenteForm.get('date')?.value) {
      const filter: MargeVenteFilter = {
        annee: this.margeVenteForm.get('annee')?.value,
        // mois: this.margeVenteForm.get('mois')?.value,
        date: this.margeVenteForm.get('date')?.value,
        depot: this.margeVenteForm.get('depot')?.value,
        debut: new Date,
        fin: new Date,
        position: this.position
      };

      this.margeVenteService.genererRapport(filter)
        .pipe(takeUntil(this.destroy$))
        .subscribe({
          next: (blob) => {
            const url = window.URL.createObjectURL(blob);
            const link = document.createElement('a');
            link.href = url;
            link.download = `rapport-marge-${new Date().toISOString().split('T')[0]}.pdf`;
            link.click();
            window.URL.revokeObjectURL(url);
          },
          error: (error) => {
            console.error('Erreur lors de la génération du rapport:', error);
          }
        });
    }
  }

  }

    if (this.selectedMarge && (this.selectedMarge.p?.id || this.selectedMarge.cpns?.id)) return true;
    return false;

    return this.position === 2 && this.margesVente.length > 0;
  }
